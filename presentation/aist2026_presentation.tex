% SLIDE 3: THE LINGUISTIC BOTTLENECK
% ==========================================
\begin{frame}{The Linguistic Bottleneck: Agglutinative Morphology}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \subhead{Kazakh Morphological Typology}
            \begin{itemize}\setlength{\itemsep}{4pt}
                \item \textbf{Agglutinative structure}: Extensive inflectional suffixes attach to invariant root stems.
                \item \textbf{Canonical suffix order}: \\
                \centerline{\small $\text{Stem} \to \text{Derivation} \to \text{Plural} \to \text{Possessive} \to \text{Case}$}
                \item \textbf{Clausal compounding}: Single inflected words convey complete clauses (e.g., \textit{жаз-ба-лар-ыңыз-дан} = write - [nom] - [plural] - [your] - [from/abl]), expanding lexical surface space exponentially.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \subhead{Subword Tokenizer Inflation}
            \begin{itemize}\setlength{\itemsep}{3pt}
                \item Frequency-based tokenizers (WordPiece, BPE) slice without grammatical awareness.
                \item Diagnostic example: \textit{«Қазақстандықтардың»} \color{SlateText}\footnotesize(of the people of Kazakhstan):
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \color{AlertRed}\textbf{Standard WordPiece}: \texttt{\footnotesize[Қа, за, қс, тан, дық, тар, дың]} (7 subword fragments)
                    \item \color{AccentGreen}\textbf{Hybrid FST decomposition}: \texttt{\footnotesize[Қазақ, -стан, -дық, -тар, -дың]} (stem + morphemes)
                \end{itemize}
                \item Uncontrolled subword fragmentation inflates token counts by up to \textbf{+57\%}, diluting semantic representations.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{The Operational Hazard: Short Reviews Trigger False Accusations}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \subhead{Asymmetric Penalty of False Alarms}
            \begin{itemize}\setlength{\itemsep}{5pt}
                \item In e-commerce auditing and education, false accusations inflict irreversible reputation and commercial damage.
                \item Consumer feedback in mobile applications is naturally brief: over 45\% of reviews contain $\le 10$ words ($N=1,842$ short reviews).
                \item Extreme context scarcity starves self-attention, skewing standard likelihood heuristics toward false alarms.
            \end{itemize}
        \end{column}

            {\small\color{SlateText}Does dedicated pretraining on conversational Kazakh (\textbf{KazBERT}, \textbf{KazRoBERTa}) provide superior inductive bias over multilingual foundations (\textbf{mBERT}, \textbf{XLM-R}) for noisy consumer reviews?}

            \subhead{RQ2: Normalization}
            \textbf{\color{DarkNavy}Inflation \& Convergence}\\[4pt]
            {\small\color{SlateText}Can a rule-based Finite-State Transducer (\textbf{FST}) pre-segmentation universally bound token inflation and establish a stable morphemic convergence limit across tokenizers?}
        \end{column}
        \begin{column}{0.31\textwidth}
            \subhead{RQ3: Mitigation}
            \textbf{\color{DarkNavy}False Positive Defense}\\[4pt]
            {\small\color{SlateText}Does morphological grounding eliminate short-text false accusations, and does this robustness generalize to out-of-distribution (\textbf{OOD}) genres?}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Proposed Method: Hybrid FST Morphological Analyzer}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{accentcard}[height=5.4cm]{Three-Tier Linguistic Engine}
                \scriptsize
                \textbf{1. Tier 1: Nominal Inflectional Chains}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Plurals (\textit{-лар, -лер, -дар, -дер, -тар, -тер})
                    \item Possessive markers (\textit{-ымыз, -іңіз, -сы, -сі})
                    \item 7 Case endings (\textit{-ның, -ге, -ны, -дан, -мен})
                \end{itemize}
                \vspace{2pt}
                \textbf{2. Tier 2: Verbal Inflections \& Participles}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Participles: \textit{-ғандықтан, -ген, -атын, -йді}
                    \item Person / agreement: \textit{-мын, -сің, -мыз}
                \end{itemize}
                \vspace{2pt}
                \textbf{3. Tier 3: Code-Switched Loanword Handling}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Strips suffixes from Russian stems: \\
                    \textit{доставка-сы}, \textit{каспи-ден}, \textit{оплата-сын}.
                \end{itemize}
            \end{accentcard}
