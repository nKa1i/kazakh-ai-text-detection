\begin{frame}{The Linguistic Bottleneck: Agglutinative Morphology}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{accentcard}[height=5.4cm]{Kazakh Morphological Typology}
                \footnotesize
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \textbf{Agglutinative Morphology}: Words are built by attaching inflectional suffixes to root stems.
                    \item \textbf{Suffix Hierarchy}: \\
                    \centerline{\scriptsize $\text{Root} \to \text{Deriv.} \to \text{Plural} \to \text{Poss.} \to \text{Case}$}
                    \item \textbf{Clausal Compounding}: Single inflected words convey complex phrases:
                    \begin{center}\scriptsize
                        \textit{жаз-ба-лар-ыңыз-дан} \\
                        (write - [nom] - [plural] - [your] - [from/abl])
                    \end{center}
                    \item Produces an exponential lexical surface space.
                \end{itemize}
            \end{accentcard}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertcard}[height=5.4cm]{Subword Tokenizer Inflation}
                \footnotesize
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item \textbf{Statistical Slicing}: Subword tokenizers (BPE, WordPiece) slice strings by frequency, destroying stems.
                \end{itemize}
                \vspace{1pt}
                \begin{mycard}[colback=white, borderline west={2pt}{0pt}{AlertRed}, top=0.6mm, bottom=0.6mm]
                    \raggedright\scriptsize
                    Target: \textbf{«Қазақстандықтардың»} \\
                    \color{AlertRed}\textbf{WordPiece}: \texttt{\tiny[Қа, за, қс, тан, дық, тар, дың]} \\
                    \color{AccentGreen}\textbf{Hybrid FST}: \texttt{\tiny[Қазақ, -стан, -дық, -тар, -дың]}
                \end{mycard}
                \vspace{2pt}
                \begin{itemize}\setlength{\itemsep}{1pt}\footnotesize
                    \item \textbf{Attention Dilution}: Inflates token counts by up to \textbf{+57\%}, diluting transformer representations.
                \end{itemize}
            \end{alertcard}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Proposed Method: Hybrid FST Morphological Analyzer}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{accentcard}[height=5.4cm]{Three-Tier Linguistic Engine}
                \scriptsize
                \textbf{1. Tier 1: Nominal Inflectional Chains}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Plurals (\textit{-лар, -лер, -дар, -дер, -тар, -тер})
                    \item Possessive markers (\textit{-ымыз, -іңіз, -сы, -сі})
                    \item 7 Case endings (\textit{-ның, -ге, -ны, -дан, -мен})
                \end{itemize}
                \vspace{2pt}
                \textbf{2. Tier 2: Verbal Inflections \& Participles}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Participles: \textit{-ғандықтан, -ген, -атын, -йді}
                    \item Person / agreement: \textit{-мын, -сің, -мыз}
                \end{itemize}
                \vspace{2pt}
                \textbf{3. Tier 3: Code-Switched Loanword Handling}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Strips suffixes from Russian stems: \\
                    \textit{доставка-сы}, \textit{каспи-ден}, \textit{оплата-сын}.
                \end{itemize}
            \end{accentcard}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{greencard}[height=5.4cm]{Morphological Pre-Segmentation}
                \scriptsize
                \textbf{Standard Native Review}: \\
                {\scriptsize \textit{«Бұл жазбаларыңыздан жасалғандықтан...»}} \\[1pt]
                \centering\color{SlateText} $\downarrow$ \textbf{Hybrid FST Decomposition} $\downarrow$ \\[1pt]
                {\scriptsize\color{DarkNavy}\textbf{«Бұл жазба {\color{TechCyan}-лар -ыңыз -дан} жаса {\color{TechCyan}-л -ғандықтан}...»}} \par
                \vspace{4pt}
                \raggedright\textbf{Code-Switched Review}: \\
                {\scriptsize \textit{«Каспиден доставкасы өте тез болды...»}} \\[1pt]
                \centering\color{SlateText} $\downarrow$ \textbf{Hybrid FST Decomposition} $\downarrow$ \\[1pt]
                {\scriptsize\color{DarkNavy}\textbf{«Каспи {\color{TechCyan}-ден} доставка {\color{TechCyan}-сы} өте тез бол {\color{TechCyan}-ды}...»}} \par
                \vspace{3pt}
                \raggedright

        \end{column}
    \end{columns}

                \end{tcolorbox}

                    \item \textbf{McNemar's Paired Test}:
                    $\chi^2 = 21.04, \quad p = 0.000004 \quad (p < 10^{-5})$ \\

                    \item \textbf{Bootstrap Resampling ($B=1,000$ iterations)}:
                    \begin{itemize}\setlength{\itemsep}{0pt}\scriptsize
                        \item Pure FPs 95\% CI: $[39.0, 68.0]$
                        \item Hybrid FST FPs 95\% CI: $[20.0, 41.0]$
                        \item Zero overlap confirms deterministic drop.
                    \end{itemize}
                \end{itemize}
            \end{accentcard}
